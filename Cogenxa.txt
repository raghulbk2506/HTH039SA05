const float MIN_WATER_PER_SESSION = 0.02;

// ============================================================
// SETUP
// ============================================================

void setup() {

  Serial.begin(115200);

  delay(500);

  Serial.println();
  Serial.println("======================================");
  Serial.println("      AQUASHIELD SMART IRRIGATION");
  Serial.println("======================================");

  // ----------------------------------------------------------
  // GPIO
  // ----------------------------------------------------------

  setupPins();


  // ----------------------------------------------------------
  // FLOW SENSOR INTERRUPT
  // ----------------------------------------------------------

  attachInterrupt(
    digitalPinToInterrupt(FLOW_PIN),
    flowPulse,
    FALLING
  );


  // ----------------------------------------------------------
  // DHT
  // ----------------------------------------------------------

  dht.begin();


  // ----------------------------------------------------------
  // OLED
  // ----------------------------------------------------------

  Wire.begin(21, 22);

  if (!display.begin(
        SSD1306_SWITCHCAPVCC,
        OLED_ADDRESS
      )) {

    Serial.println("OLED not detected!");

  } else {

    Serial.println("OLED OK");

    display.clearDisplay();

    display.setTextColor(SSD1306_WHITE);

    display.setTextSize(1);

    display.setCursor(0, 0);
    display.println("AquaShield");

    display.setCursor(0, 15);
    display.println("Smart Irrigation");

    display.setCursor(0, 30);
    display.println("Initializing...");

    display.display();

    delay(2000);
  }


  // ----------------------------------------------------------
  // INITIALIZE VALUES
  // ----------------------------------------------------------

  closeAllValves();

  digitalWrite(WARNING_LED, LOW);

  digitalWrite(BUZZER_PIN, LOW);

  setZoneLEDs(-1);


  // ----------------------------------------------------------
  // INITIAL SENSOR READ
  // ----------------------------------------------------------

  readSensors();

  Serial.println("System ready.");
  Serial.println();
}


// ============================================================
// MAIN LOOP
// ============================================================

void loop() {

  unsigned long now = millis();


  // ----------------------------------------------------------
  // READ SENSORS
  // ----------------------------------------------------------

  if (now - lastSensorRead >= SENSOR_INTERVAL) {

    lastSensorRead = now;

    readSensors();

    calculateWaterUsed();

    printStatus();
  }


  // ----------------------------------------------------------
  // DECISION ENGINE
  // ----------------------------------------------------------

  if (now - lastDecision >= DECISION_INTERVAL) {

    lastDecision = now;

    if (!irrigationRunning) {

      makeIrrigationDecision();
    }
  }


  // ----------------------------------------------------------
  // IRRIGATION SAFETY CHECK
  // ----------------------------------------------------------

  if (irrigationRunning) {

    // Maximum irrigation time
    if (millis() - irrigationStartTime >
        MAX_IRRIGATION_TIME) {

      Serial.println(
        "Maximum irrigation time reached."
      );

      stopIrrigation();
    }


    // Tank too low
    if (tankPercent < MIN_TANK_PERCENT) {

      Serial.println(
        "Tank level too low!"
      );

      stopIrrigation();

      digitalWrite(WARNING_LED, HIGH);

      warningBeep();
    }


    // Quota exhausted
    if (remainingQuota <= 0.0) {

      Serial.println(
        "Water quota exhausted!"
      );

      stopIrrigation();

      digitalWrite(WARNING_LED, HIGH);

      warningBeep();
    }


    // Rain detected
    if (raining) {

      Serial.println(
        "Rain detected - stopping irrigation."
      );

      stopIrrigation();
    }
  }


  // ----------------------------------------------------------
  // STOP ZONE WHEN MOISTURE TARGET REACHED
  // ----------------------------------------------------------

  if (irrigationRunning &&
      activeZone >= 0) {

    if (soilPercent[activeZone] >= SOIL_TARGET) {

      Serial.println(
        "Target moisture reached."
      );

      stopIrrigation();
    }
  }


  // ----------------------------------------------------------
  // OLED UPDATE
  // ----------------------------------------------------------

  if (now - lastDisplayUpdate >= DISPLAY_INTERVAL) {

    lastDisplayUpdate = now;

    updateDisplay();
  }
}


// ============================================================
// GPIO SETUP
// ============================================================

void setupPins() {

  // Relays
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);
  pinMode(RELAY3_PIN, OUTPUT);


  // LEDs
  pinMode(LED_ZONE1, OUTPUT);
  pinMode(LED_ZONE2, OUTPUT);
  pinMode(LED_ZONE3, OUTPUT);

  pinMode(WARNING_LED, OUTPUT);


  // Buzzer
  pinMode(BUZZER_PIN, OUTPUT);


  // Flow
  pinMode(FLOW_PIN, INPUT_PULLUP);


  // Ultrasonic
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);


  // Rain
  if (RAIN_SENSOR_ENABLED) {

    pinMode(RAIN_PIN, INPUT_PULLUP);
  }


  // Turn everything off
  closeAllValves();

  setZoneLEDs(-1);

  digitalWrite(WARNING_LED, LOW);
  digitalWrite(BUZZER_PIN, LOW);
}


// ============================================================
// FLOW INTERRUPT
// ============================================================

void IRAM_ATTR flowPulse() {

  flowPulses++;
}


// ============================================================
// READ ALL SENSORS
// ============================================================

void readSensors() {

  readSoilSensors();

  readDHT();

  readTankLevel();

  readRainSensor();
}


// ============================================================
// SOIL SENSOR
// ============================================================

void readSoilSensors() {

  soilRaw[0] = analogRead(SOIL1_PIN);
  soilRaw[1] = analogRead(SOIL2_PIN);
  soilRaw[2] = analogRead(SOIL3_PIN);


  /*
     Capacitive soil sensors normally produce different
     raw values depending on their calibration.

     The values below are STARTING values only.

     You should calibrate using:
       dry soil
       wet soil

     If your sensor behaves opposite to this mapping,
     reverse the values.
  */

  for (int i = 0; i < 3; i++) {

    soilPercent[i] =
      map(
        soilRaw[i],
        3000,
        1200,
        0,
        100
      );

    soilPercent[i] =
      constrain(
        soilPercent[i],
        0,
        100
      );
  }
}


// ============================================================
// DHT11
// ============================================================

void readDHT() {

  float h = dht.readHumidity();

  float t = dht.readTemperature();


  if (!isnan(h)) {

    humidity = h;
  }


  if (!isnan(t)) {

    temperature = t;
  }
}


// ============================================================
// ULTRASONIC TANK LEVEL
// ============================================================

void readTankLevel() {

  digitalWrite(TRIG_PIN, LOW);

  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);

  delayMicroseconds(10);

  digitalWrite(TRIG_PIN, LOW);


  unsigned long duration =
    pulseIn(
      ECHO_PIN,
      HIGH,
      30000
    );


  if (duration == 0) {

    Serial.println(
      "Ultrasonic timeout."
    );

    return;
  }


  tankDistance =
    duration * 0.0343 / 2.0;


  /*
     Example:

     Tank height = 18 cm
     Sensor mounted at top

     distance small -> tank full
     distance large -> tank empty
  */

  float waterHeight =
    TANK_HEIGHT_CM - tankDistance;


  tankPercent =
    (waterHeight / TANK_HEIGHT_CM) * 100.0;


  tankPercent =
    constrain(
      tankPercent,
      0,
      100
    );
}


// ============================================================
// RAIN SENSOR
// ============================================================

void readRainSensor() {

  if (!RAIN_SENSOR_ENABLED) {

    raining = false;

    return;
  }


  /*
     THIS PART DEPENDS ON YOUR RAIN SENSOR INTERFACE.

     For a digital rain module where LOW means rain:

     raining = (digitalRead(RAIN_PIN) == LOW);

     Do not enable this until your two-pin sensor has
     been correctly interfaced.
  */

  raining =
    (digitalRead(RAIN_PIN) == LOW);
}


// ============================================================
// CALCULATE FLOW / WATER USED
// ============================================================

void calculateWaterUsed() {

  noInterrupts();

  unsigned long pulses =
    flowPulses;

  flowPulses = 0;

  interrupts();


  if (pulses == 0) {

    return;
  }


  float liters =
    (float)pulses /
    FLOW_PULSES_PER_LITER;


  totalWaterLiters += liters;

  irrigationSessionLiters += liters;


  if (irrigationRunning &&
      activeZone >= 0) {

    zoneWaterUsed[activeZone] += liters;

    remainingQuota -= liters;

    if (remainingQuota < 0) {

      remainingQuota = 0;
    }
  }
}


// ============================================================
// IRRIGATION DECISION ENGINE
// ============================================================

void makeIrrigationDecision() {

  // ----------------------------------------------------------
  // HARD QUOTA
  // ----------------------------------------------------------

  if (remainingQuota <= 0.0) {

    Serial.println(
      "DECISION: Quota exhausted."
    );

    digitalWrite(WARNING_LED, HIGH);

    return;
  }


  // ----------------------------------------------------------
  // TANK SAFETY
  // ----------------------------------------------------------

  if (tankPercent < MIN_TANK_PERCENT) {

    Serial.println(
      "DECISION: Tank level too low."
    );

    digitalWrite(WARNING_LED, HIGH);

    warningBeep();

    return;
  }


  // ----------------------------------------------------------
  // RAIN
  // ----------------------------------------------------------

  if (raining) {

    Serial.println(
      "DECISION: Rain detected."
    );

    return;
  }


  // ----------------------------------------------------------
  // FIND HIGHEST PRIORITY ZONE
  // ----------------------------------------------------------

  int selectedZone =
    findHighestPriorityZone();


  if (selectedZone < 0) {

    Serial.println(
      "DECISION: No zone needs irrigation."
    );

    digitalWrite(WARNING_LED, LOW);

    return;
  }


  // ----------------------------------------------------------
  // START IRRIGATION
  // ----------------------------------------------------------

  startIrrigation(selectedZone);
}


// ============================================================
// CALCULATE ZONE PRIORITY
// ============================================================

int calculatePriority(int zone) {

  int priority = 0;


  // ----------------------------------------------------------
  // DRYNESS PRIORITY
  // ----------------------------------------------------------

  if (soilPercent[zone] < 30) {

    priority += 60;

  } else if (soilPercent[zone] < 40) {

    priority += 45;

  } else if (soilPercent[zone] < 50) {

    priority += 25;

  } else {

    priority += 0;
  }


  // ----------------------------------------------------------
  // TEMPERATURE STRESS
  // ----------------------------------------------------------

  if (temperature >= 35) {

    priority += 20;

  } else if (temperature >= 32) {

    priority += 10;
  }


  // ----------------------------------------------------------
  // LOW HUMIDITY
  // ----------------------------------------------------------

  if (humidity < 40) {

    priority += 10;

  } else if (humidity < 55) {

    priority += 5;
  }


  // ----------------------------------------------------------
  // PRE-EMPTIVE WATERING DEMO RULE
  // ----------------------------------------------------------

  /*
     This is NOT a real weather forecast.

     It demonstrates the hackathon concept of
     pre-emptive watering before heat/dry conditions.

     Later this can be replaced by actual weather API
     forecast data.
  */

  if (temperature >= 34 &&
      humidity <= 45) {

    priority += 15;
  }


  // ----------------------------------------------------------
  // PREVENT ALREADY-WET ZONES
  // ----------------------------------------------------------

  if (soilPercent[zone] >= SOIL_TARGET) {

    priority = 0;
  }


  return priority;
}


// ============================================================
// FIND HIGHEST PRIORITY ZONE
// ============================================================

int findHighestPriorityZone() {

  int bestZone = -1;

  int bestPriority = 0;


  for (int i = 0; i < 3; i++) {

    int p =
      calculatePriority(i);


    Serial.print(
      "Zone "
    );

    Serial.print(i + 1);

    Serial.print(
      " priority = "
    );

    Serial.println(p);


    if (p > bestPriority) {

      bestPriority = p;

      bestZone = i;
    }
  }


  return bestZone;
}


// ============================================================
// START IRRIGATION
// ============================================================

void startIrrigation(int zone) {

  if (zone < 0 ||
      zone > 2) {

    return;
  }


  if (remainingQuota <= 0) {

    return;
  }


  if (tankPercent < MIN_TANK_PERCENT) {

    return;
  }


  // ----------------------------------------------------------
  // CLOSE EVERYTHING FIRST
  // ----------------------------------------------------------

  closeAllValves();


  // ----------------------------------------------------------
  // SELECT ZONE
  // ----------------------------------------------------------

  activeZone = zone;

  irrigationRunning = true;

  irrigationStartTime =
    millis();

  irrigationSessionLiters = 0;


  // ----------------------------------------------------------
  // OPEN SELECTED VALVE
  // ----------------------------------------------------------

  openValve(zone);

  setZoneLEDs(zone);


  // ----------------------------------------------------------
  // DISPLAY / SERIAL
  // ----------------------------------------------------------

  Serial.println();
  Serial.println(
    "================================"
  );

  Serial.print(
    "IRRIGATION STARTED - ZONE "
  );

  Serial.println(
    zone + 1
  );

  Serial.print(
    "Soil = "
  );

  Serial.print(
    soilPercent[zone]
  );

  Serial.println("%");


  Serial.print(
    "Priority = "
  );

  Serial.println(
    calculatePriority(zone)
  );


  Serial.print(
    "Remaining quota = "
  );

  Serial.print(
    remainingQuota,
    3
  );

  Serial.println(" L");


  Serial.println(
    "Pump should be running."
  );

  Serial.println(
    "================================"
  );


  irrigationCount[zone]++;

  alarmBeep();
}


// ============================================================
// STOP IRRIGATION
// ============================================================

void stopIrrigation() {

  if (!irrigationRunning) {

    closeAllValves();

    setZoneLEDs(-1);

    return;
  }


  closeAllValves();

  setZoneLEDs(-1);


  Serial.println();
  Serial.println(
    "--------------------------------"
  );

  Serial.print(
    "IRRIGATION STOPPED - ZONE "
  );

  Serial.println(
    activeZone + 1
  );


  Serial.print(
    "Water used this session: "
  );

  Serial.print(
    irrigationSessionLiters,
    3
  );

  Serial.println(" L");


  Serial.print(
    "Remaining quota: "
  );

  Serial.print(
    remainingQuota,
    3
  );

  Serial.println(" L");


  Serial.println(
    "--------------------------------"
  );


  irrigationRunning = false;

  activeZone = -1;

  irrigationSessionLiters = 0;
}


// ============================================================
// OPEN ONE VALVE
// ============================================================

void openValve(int zone) {

  closeAllValves();


  if (zone == 0) {

    setRelay(
      RELAY1_PIN,
      true
    );

  } else if (zone == 1) {

    setRelay(
      RELAY2_PIN,
      true
    );

  } else if (zone == 2) {

    setRelay(
      RELAY3_PIN,
      true
    );
  }
}


// ============================================================
// CLOSE ALL VALVES
// ============================================================

void closeAllValves() {

  setRelay(
    RELAY1_PIN,
    false
  );

  setRelay(
    RELAY2_PIN,
    false
  );

  setRelay(
    RELAY3_PIN,
    false
  );
}


// ============================================================
// RELAY CONTROL
// ============================================================

void setRelay(
  int relayPin,
  bool state
) {

  if (RELAY_ACTIVE_LOW) {

    digitalWrite(
      relayPin,
      state ? LOW : HIGH
    );

  } else {

    digitalWrite(
      relayPin,
      state ? HIGH : LOW
    );
  }
}


// ============================================================
// ZONE LEDS
// ============================================================

void setZoneLEDs(int zone) {

  digitalWrite(
    LED_ZONE1,
    LOW
  );

  digitalWrite(
    LED_ZONE2,
    LOW
  );

  digitalWrite(
    LED_ZONE3,
    LOW
  );


  if (zone == 0) {

    digitalWrite(
      LED_ZONE1,
      HIGH
    );

  } else if (zone == 1) {

    digitalWrite(
      LED_ZONE2,
      HIGH
    );

  } else if (zone == 2) {

    digitalWrite(
      LED_ZONE3,
      HIGH
    );
  }
}


// ============================================================
// OLED UPDATE
// ============================================================

void updateDisplay() {

  if (!display.width()) {

    return;
  }


  display.clearDisplay();

  display.setTextColor(
    SSD1306_WHITE
  );

  display.setTextSize(1);


  // ----------------------------------------------------------
  // LINE 1
  // ----------------------------------------------------------

  display.setCursor(0, 0);

  display.print("AquaShield");


  // ----------------------------------------------------------
  // LINE 2
  // ----------------------------------------------------------

  display.setCursor(0, 10);

  display.print("Z1:");
  display.print(soilPercent[0]);

  display.print("% Z2:");
  display.print(soilPercent[1]);

  display.print("%");


  // ----------------------------------------------------------
  // LINE 3
  // ----------------------------------------------------------

  display.setCursor(0, 20);

  display.print("Z3:");
  display.print(soilPercent[2]);

  display.print("% T:");
  display.print((int)temperature);

  display.print("C");


  // ----------------------------------------------------------
  // LINE 4
  // ----------------------------------------------------------

  display.setCursor(0, 30);

  display.print("Tank:");
  display.print((int)tankPercent);

  display.print("%");


  // ----------------------------------------------------------
  // LINE 5
  // ----------------------------------------------------------

  display.setCursor(0, 40);

  display.print("Quota:");
  display.print(remainingQuota, 2);

  display.print("L");


  // ----------------------------------------------------------
  // LINE 6
  // ----------------------------------------------------------

  display.setCursor(0, 50);

  if (irrigationRunning) {

    display.print("IRRIGATING Z");
    display.print(activeZone + 1);

  } else {

    display.print("System: READY");
  }


  display.display();
}

    BUZZER_PIN,
    LOW

}


// ============================================================
// WARNING BEEP
// ============================================================

void warningBeep() {

  for (int i = 0; i < 2; i++) {

    digitalWrite(
      BUZZER_PIN,
      HIGH
    );

    delay(150);

    digitalWrite(
      BUZZER_PIN,
      LOW
    );

    delay(100);
  }
}
